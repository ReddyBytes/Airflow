# S3 Operators — Code Examples

## Example 1: Upload Local File to S3

Upload a file generated by your pipeline to an S3 bucket.

```python
from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.operators.bash import BashOperator
from airflow.providers.amazon.aws.transfers.local_to_s3 import LocalFilesystemToS3Operator
from airflow.providers.amazon.aws.operators.s3 import S3CreateBucketOperator

BUCKET_NAME = "my-company-data-pipeline"
AWS_CONN_ID = "aws_default"

def generate_csv_report(**context):
    """Generate a daily CSV report locally before uploading to S3."""
    import csv
    import os

    date = context["ds"]
    output_dir = f"/tmp/pipeline/reports/{date}"
    os.makedirs(output_dir, exist_ok=True)
    output_file = f"{output_dir}/daily_summary.csv"

    # Simulate generating a report
    rows = [
        {"date": date, "product": "Widget A", "quantity": 150, "revenue": 7500.00},
        {"date": date, "product": "Widget B", "quantity": 89, "revenue": 4450.00},
        {"date": date, "product": "Gadget C", "quantity": 42, "revenue": 8400.00},
    ]

    with open(output_file, "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=["date", "product", "quantity", "revenue"])
        writer.writeheader()
        writer.writerows(rows)

    print(f"Generated report at {output_file} with {len(rows)} rows")
    return output_file


with DAG(
    dag_id="s3_upload_example",
    start_date=datetime(2024, 1, 1),
    schedule_interval="@daily",
    catchup=False,
    default_args={"retries": 2, "retry_delay": timedelta(minutes=5)},
    tags=["s3", "upload", "example"],
) as dag:

    # Step 1: Make sure bucket exists (S3CreateBucketOperator is idempotent)
    ensure_bucket = S3CreateBucketOperator(
        task_id="ensure_bucket_exists",
        bucket_name=BUCKET_NAME,
        region_name="us-east-1",
        aws_conn_id=AWS_CONN_ID,
    )

    # Step 2: Generate the local file
    generate_report = PythonOperator(
        task_id="generate_daily_report",
        python_callable=generate_csv_report,
    )

    # Step 3: Upload to S3
    # S3 key = path inside the bucket
    upload_summary = LocalFilesystemToS3Operator(
        task_id="upload_summary_to_s3",
        filename="/tmp/pipeline/reports/{{ ds }}/daily_summary.csv",
        dest_key="reports/daily/{{ ds }}/daily_summary.csv",
        dest_bucket=BUCKET_NAME,
        aws_conn_id=AWS_CONN_ID,
        replace=True,         # Overwrite if already exists (makes it idempotent)
    )

    # Step 4: Also upload a metadata file with run info
    upload_metadata = LocalFilesystemToS3Operator(
        task_id="upload_metadata",
        filename="/tmp/pipeline/reports/{{ ds }}/metadata.json",
        dest_key="reports/daily/{{ ds }}/metadata.json",
        dest_bucket=BUCKET_NAME,
        aws_conn_id=AWS_CONN_ID,
        replace=True,
    )

    # Generate metadata file with bash
    generate_metadata = BashOperator(
        task_id="generate_metadata",
        bash_command="""
            mkdir -p /tmp/pipeline/reports/{{ ds }}
            cat > /tmp/pipeline/reports/{{ ds }}/metadata.json << EOF
{
  "date": "{{ ds }}",
  "generated_at": "$(date -u +%Y-%m-%dT%H:%M:%SZ)",
  "dag_run_id": "{{ run_id }}",
  "row_count": 3
}
EOF
        """,
    )

    ensure_bucket >> generate_report >> generate_metadata >> [upload_summary, upload_metadata]
```

## Example 2: Check if S3 Key Exists

Before processing, check whether the expected file has landed in S3. Use `S3KeySensor` to wait for it, or `S3Hook` to check programmatically.

```python
from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.providers.amazon.aws.sensors.s3 import S3KeySensor
from airflow.providers.amazon.aws.hooks.s3 import S3Hook

AWS_CONN_ID = "aws_default"
SOURCE_BUCKET = "vendor-data-drops"

def check_and_log_file_details(**context):
    """
    After sensor confirms file exists, check its properties.
    Uses S3Hook directly for more control.
    """
    ti = context["ti"]
    date = context["ds"]
    date_nodash = context["ds_nodash"]

    s3_key = f"vendor/daily/{date_nodash}/orders.csv"

    hook = S3Hook(aws_conn_id=AWS_CONN_ID)

    # Get object metadata
    obj = hook.get_key(key=s3_key, bucket_name=SOURCE_BUCKET)
    size_mb = obj.content_length / (1024 * 1024)

    print(f"File found: s3://{SOURCE_BUCKET}/{s3_key}")
    print(f"Size: {size_mb:.2f} MB")
    print(f"Last modified: {obj.last_modified}")
    print(f"ETag: {obj.e_tag}")

    # Push details to XCom for downstream tasks
    ti.xcom_push(key="file_size_mb", value=round(size_mb, 2))
    ti.xcom_push(key="s3_key", value=s3_key)

    # Warn if file is surprisingly small (< 1KB)
    if obj.content_length < 1024:
        print("WARNING: File is very small — possible empty delivery")

    return {"s3_key": s3_key, "size_mb": size_mb}


def process_if_file_exists(**context):
    """Only run if file is above minimum size threshold."""
    ti = context["ti"]
    file_size = ti.xcom_pull(task_ids="check_file_details", key="file_size_mb")
    s3_key = ti.xcom_pull(task_ids="check_file_details", key="s3_key")

    if file_size < 0.001:
        raise ValueError(f"File {s3_key} is too small ({file_size} MB) — rejecting")

    print(f"Processing {s3_key} ({file_size} MB)")
    # ... actual processing logic here


with DAG(
    dag_id="s3_key_check_example",
    start_date=datetime(2024, 1, 1),
    schedule_interval="@daily",
    catchup=False,
    tags=["s3", "sensor", "example"],
) as dag:

    # Wait for the vendor file to land in S3
    wait_for_vendor_file = S3KeySensor(
        task_id="wait_for_vendor_file",
        bucket_name=SOURCE_BUCKET,
        bucket_key="vendor/daily/{{ ds_nodash }}/orders.csv",
        aws_conn_id=AWS_CONN_ID,
        poke_interval=60,           # Check every 60 seconds
        timeout=60 * 60 * 4,        # Give up after 4 hours
        mode="reschedule",          # Release worker slot between checks
        soft_fail=False,            # Fail the task if timeout exceeded
    )

    # After file is confirmed, log its details
    check_file = PythonOperator(
        task_id="check_file_details",
        python_callable=check_and_log_file_details,
    )

    # Process it
    process_file = PythonOperator(
        task_id="process_vendor_file",
        python_callable=process_if_file_exists,
    )

    wait_for_vendor_file >> check_file >> process_file
```

## Example 3: Copy Between S3 Buckets

Move files from a staging bucket to a production bucket, with cleanup.

```python
from datetime import datetime, timedelta
from airflow import DAG
from airflow.providers.amazon.aws.operators.s3 import (
    S3CopyObjectOperator,
    S3DeleteObjectsOperator,
)
from airflow.providers.amazon.aws.hooks.s3 import S3Hook
from airflow.operators.python import PythonOperator

STAGING_BUCKET = "my-pipeline-staging"
PROD_BUCKET = "my-pipeline-production"
ARCHIVE_BUCKET = "my-pipeline-archive"
AWS_CONN_ID = "aws_default"


def list_and_copy_all_files(**context):
    """
    For cases where you need to copy multiple files with a prefix,
    use S3Hook to list keys and then copy each one.
    """
    date = context["ds"]
    date_nodash = context["ds_nodash"]
    prefix = f"processed/{date}/"

    hook = S3Hook(aws_conn_id=AWS_CONN_ID)

    # List all files under the prefix
    keys = hook.list_keys(bucket_name=STAGING_BUCKET, prefix=prefix)

    if not keys:
        print(f"No files found under s3://{STAGING_BUCKET}/{prefix}")
        return []

    print(f"Found {len(keys)} files to copy")

    copied = []
    for key in keys:
        # Construct destination key
        dest_key = key.replace(f"processed/{date}/", f"data/{date}/")

        # Copy using the hook
        hook.copy_object(
            source_bucket_key=key,
            dest_bucket_key=dest_key,
            source_bucket_name=STAGING_BUCKET,
            dest_bucket_name=PROD_BUCKET,
        )
        print(f"Copied: s3://{STAGING_BUCKET}/{key} -> s3://{PROD_BUCKET}/{dest_key}")
        copied.append(key)

    print(f"Copied {len(copied)} files to production")
    return copied


with DAG(
    dag_id="s3_copy_between_buckets",
    start_date=datetime(2024, 1, 1),
    schedule_interval="@daily",
    catchup=False,
    default_args={"retries": 1, "retry_delay": timedelta(minutes=10)},
    tags=["s3", "copy", "archive", "example"],
) as dag:

    # Copy the main output file from staging to production
    copy_main_output = S3CopyObjectOperator(
        task_id="copy_output_to_production",
        source_bucket_name=STAGING_BUCKET,
        source_bucket_key="processed/{{ ds }}/output.parquet",
        dest_bucket_name=PROD_BUCKET,
        dest_bucket_key="data/{{ ds }}/output.parquet",
        aws_conn_id=AWS_CONN_ID,
    )

    # Archive the original to archive bucket with year/month partitioning
    archive_original = S3CopyObjectOperator(
        task_id="archive_original_to_archive_bucket",
        source_bucket_name=STAGING_BUCKET,
        source_bucket_key="processed/{{ ds }}/output.parquet",
        dest_bucket_name=ARCHIVE_BUCKET,
        # Partition archive by year/month/day for efficient retrieval
        dest_bucket_key="archive/year={{ macros.ds_format(ds, '%Y-%m-%d', '%Y') }}/month={{ macros.ds_format(ds, '%Y-%m-%d', '%m') }}/day={{ macros.ds_format(ds, '%Y-%m-%d', '%d') }}/output.parquet",
        aws_conn_id=AWS_CONN_ID,
    )

    # Bulk copy multiple files with a prefix (using hook)
    copy_all_partitions = PythonOperator(
        task_id="copy_all_partition_files",
        python_callable=list_and_copy_all_files,
    )

    # After promoting to production, delete from staging (save storage costs)
    cleanup_staging = S3DeleteObjectsOperator(
        task_id="delete_from_staging",
        bucket=STAGING_BUCKET,
        keys=[
            "processed/{{ ds }}/output.parquet",
            "processed/{{ ds }}/metadata.json",
        ],
        aws_conn_id=AWS_CONN_ID,
    )

    # Run copy and archive in parallel, then clean up
    [copy_main_output, archive_original, copy_all_partitions] >> cleanup_staging
```

**What to notice:**
- `S3CopyObjectOperator` copies a single object between keys/buckets — no local disk needed
- `S3DeleteObjectsOperator` takes a list of keys to delete
- `S3Hook` gives you lower-level access for bulk operations (list, copy many, etc.)
- Use `replace=True` on uploads to make the DAG safe to re-run
- S3 keys are like file paths — use prefixes (`processed/{{ ds }}/`) to organize by date
